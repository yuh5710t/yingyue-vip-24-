-- ================= 验证逻辑 =================
-- 判断名字在不在白名单里的函数
local function isNameInWhitelist(name)
    for _, v in ipairs(whitelist) do
        if v == name then return true end
    end
    return false
end

btn.MouseButton1Click:Connect(function()

    task.wait(0.5)
    
    -- 1. 自动检测玩家名字是否在白名单里
    if isNameInWhitelist(player.Name) then
        status.Text = "✅ 验证通过 (白名单用户)"
        status.TextColor3 = Color3.fromRGB(0, 180, 50)
        btn.BackgroundColor3 = Color3.fromRGB(0, 200, 50)

        task.wait(1)

        -- 👇【这里放你验证成功后的主界面加载代码】
        -- loadstring(game:HttpGet("你的主界面链接"))()
        return
    end
    
    -- 2. 验证卡密 (你自己加的卡密写在这里)
    local inputText = input.Text
    if inputText == "YY-666" or inputText == "SVIP" then 
        status.Text = "✅ 验证通过 (卡密用户)"

        return

    -- 3. 失败
    btn.Text = "重新验证"
    btn.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
    status.Text = "❌ 验证失败：你不在白名单内或卡密错误"
    status.TextColor3 = Color3.fromRGB(200, 50, 50)
    task.wait(1.5)
