-- ==========================================
-- 验证逻辑：自动检测名字 + 卡密兜底
-- ==========================================
local function checkViaWorker(inputText)
    -- 如果输入为空，直接用玩家名字
    local keyToUse = inputText
    if keyToUse == "" then
        keyToUse = player.Name
    end
    
    -- 同时把 keyToUse 和真实名字传给后端
    local url = workerUrl .. "/verify?key=" .. HttpService:UrlEncode(keyToUse) .. "&name=" .. HttpService:UrlEncode(player.Name)
    local ok, res = pcall(function() return game:HttpGet(url) end)
    if not ok or not res then return nil end
    local ok2, data = pcall(function() return HttpService:JSONDecode(res) end)
    if not ok2 then return nil end
    return data
end

local function checkAuth(inputText)
    -- 1. 先按规则验证
    local w = checkViaWorker(inputText)
    if w and w.ok and w.token then 
        return true, "", w.token 
    end
    
    -- 2. 如果失败了，但玩家填了东西，强制用他的名字再查一次
    if inputText ~= "" then
        local w2 = checkViaWorker("") 
        if w2 and w2.ok and w2.token then 
            return true, "", w2.token 
        end
    end
    
    -- 3. 彻底失败
    return false, (w and w.msg) or "卡密错误", nil
end

        _G.__shadowmoon_token = token

            return game:HttpGet(workerUrl .. "/ui?token=" .. HttpService:UrlEncode(token))
        end)

            print("UI 拉取失败，token 可能过期")
        end
    else

        local reason = (msg ~= "" and msg) or "卡密错误"
